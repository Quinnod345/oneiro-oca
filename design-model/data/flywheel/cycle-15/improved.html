23 /// Sends a typed request and decodes the response.
24 @discardableResult
25 public func send<R: Request>(_ req: R) async throws -> R.Response {
26 var attempt = 0
27 repeat {
28 do {
29 let urlReq = try buildURLRequest(from: req)
30 let (data, response) = try await session.data(for: urlReq)
31 try validate(response)
32 return try decoder.decode(R.Response.self, from: data)
33 } catch let err as APIError where err.isRetryable {
34 guard attempt < retryPolicy.maxAttempts else { throw err }
35 try await Task.sleep(for: retryPolicy.delay(attempt: attempt))
36 attempt += 1
37 }
38 } while true
39 }
